1. Objective

Genomic prediction models require numerical matrices that describe fixed effects and genetic similarity among individuals.

In this step, the SSR genotypes are converted into allele dosages, the fixed effects of block and family are represented in a design matrix, and five cross-validation folds are defined. Marker processing is repeated using only the training individuals of each fold.

3. Allele-dosage coding

Each SSR code contains two allele labels. The code 12, for example, represents one copy of allele 1 and one copy of allele 2. The code 22 represents two copies of allele 2.

One column is created for every allele observed at each retained locus. The dosage is 0, 1, or 2 according to the number of copies carried by an individual.

5. Complete marker matrix

Missing allele dosages are replaced by the mean dosage of their column. The columns are then centered by subtracting those means.

The resulting matrix \(\mathbf{M}\) has individuals in rows and retained allele dosages in columns.

marker_means_full <- colMeans(
  dosage_full,
  na.rm = TRUE
)

dosage_full_imputed <- dosage_full

for (j in seq_len(ncol(dosage_full_imputed))) {
  missing_values <- is.na(
    dosage_full_imputed[, j]
  )

  dosage_full_imputed[
    missing_values,
    j
  ] <- marker_means_full[j]
}

M_full <- sweep(
  dosage_full_imputed,
  2,
  marker_means_full,
  FUN = "-"
)

6. Complete genomic relationship matrix

Let \(p_j\) be the frequency of the allele represented by column \(j\). The additive genomic relationship matrix is calculated as

\[ \mathbf{G} = \frac{\mathbf{M}\mathbf{M}^{\mathsf{T}}} {\sum_j 2p_j(1-p_j)}. \]

allele_frequencies_full <-
  marker_means_full / 2

denominator_full <- sum(
  2 *
    allele_frequencies_full *
    (1 - allele_frequencies_full)
)

G_full <- tcrossprod(
  M_full
) / denominator_full

G_full <- (
  G_full + t(G_full)
) / 2

dimnames(G_full) <- list(
  ids,
  ids
)

eigenvalues_G_full <- eigen(
  G_full,
  symmetric = TRUE,
  only.values = TRUE
)$values

9. Training-specific matrices

For each fold, locus filtering, allele coding, allele-frequency control, imputation, centering, and the genomic relationship matrix are recalculated using the 120 training individuals.

The 30 validation individuals are transformed with the means and allele structure obtained from training.

cv_objects <- vector(
  "list",
  N_FOLDS
)

fold_diagnostics <- vector(
  "list",
  N_FOLDS
)

for (fold in seq_len(N_FOLDS)) {
  validation_index <- which(
    folds == fold
  )

  training_index <- which(
    folds != fold
  )

  ssr_training <- ssr_raw[
    training_index,
    ,
    drop = FALSE
  ]

  genotype_classes <- integer(
    ncol(ssr_training)
  )

  for (j in seq_len(ncol(ssr_training))) {
    genotype_classes[j] <- length(
      unique(
        na.omit(
          ssr_training[, j]
        )
      )
    )
  }

  locus_audit <- data.frame(
    Locus = colnames(ssr_training),
    N_called = colSums(!is.na(ssr_training)),
    Call_rate = colMeans(!is.na(ssr_training)),
    N_genotype_classes = genotype_classes
  )

  locus_audit$Keep_locus <-
    locus_audit$Call_rate >= CALL_RATE_MIN &
    locus_audit$N_genotype_classes >= 2

  retained_loci <- locus_audit$Locus[
    locus_audit$Keep_locus
  ]

  ssr_fold <- ssr_raw[
    ,
    retained_loci,
    drop = FALSE
  ]

  dosage_parts <- list()
  allele_map_parts <- list()
  marker_counter <- 0L

  for (j in seq_len(ncol(ssr_fold))) {
    codes_all <- ssr_fold[, j]
    codes_training <- ssr_fold[
      training_index,
      j
    ]

    allele_1_training <- floor(
      codes_training / 10
    )

    allele_2_training <-
      codes_training %% 10

    observed_alleles <- sort(
      unique(
        c(
          allele_1_training[
            !is.na(codes_training)
          ],
          allele_2_training[
            !is.na(codes_training)
          ]
        )
      )
    )

    allele_1_all <- floor(
      codes_all / 10
    )

    allele_2_all <-
      codes_all %% 10

    for (allele in observed_alleles) {
      marker_counter <- marker_counter + 1L

      dosage <-
        as.numeric(
          allele_1_all == allele
        ) +
        as.numeric(
          allele_2_all == allele
        )

      dosage[
        is.na(codes_all)
      ] <- NA_real_

      marker_name <- paste0(
        colnames(ssr_fold)[j],
        "__A",
        allele
      )

      dosage_parts[[marker_counter]] <-
        dosage

      allele_map_parts[[marker_counter]] <-
        data.frame(
          Locus = colnames(ssr_fold)[j],
          Allele = allele,
          Marker = marker_name
        )
    }
  }

  dosage_raw <- do.call(
    cbind,
    dosage_parts
  )

  allele_audit <- bind_rows(
    allele_map_parts
  )

  rownames(dosage_raw) <- ids
  colnames(dosage_raw) <-
    allele_audit$Marker

  allele_audit$Allele_frequency <-
    colMeans(
      dosage_raw[
        training_index,
        ,
        drop = FALSE
      ],
      na.rm = TRUE
    ) / 2

  dosage_sd <- numeric(
    ncol(dosage_raw)
  )

  for (j in seq_len(ncol(dosage_raw))) {
    dosage_sd[j] <- sd(
      dosage_raw[
        training_index,
        j
      ],
      na.rm = TRUE
    )
  }

  allele_audit$Dosage_sd <-
    dosage_sd

  allele_audit$Keep_allele <-
    is.finite(
      allele_audit$Allele_frequency
    ) &
    allele_audit$Allele_frequency >=
      ALLELE_FREQUENCY_MIN &
    allele_audit$Allele_frequency <=
      (1 - ALLELE_FREQUENCY_MIN) &
    is.finite(
      allele_audit$Dosage_sd
    ) &
    allele_audit$Dosage_sd > 0

  retained_markers <-
    allele_audit$Marker[
      allele_audit$Keep_allele
    ]

  dosage <- dosage_raw[
    ,
    retained_markers,
    drop = FALSE
  ]

  marker_means <- colMeans(
    dosage[
      training_index,
      ,
      drop = FALSE
    ],
    na.rm = TRUE
  )

  dosage_imputed <- dosage

  for (j in seq_len(ncol(dosage_imputed))) {
    missing_values <- is.na(
      dosage_imputed[, j]
    )

    dosage_imputed[
      missing_values,
      j
    ] <- marker_means[j]
  }

  M <- sweep(
    dosage_imputed,
    2,
    marker_means,
    FUN = "-"
  )

  allele_frequencies <-
    marker_means / 2

  denominator <- sum(
    2 *
      allele_frequencies *
      (1 - allele_frequencies)
  )

  G <- tcrossprod(
    M
  ) / denominator

  G <- (
    G + t(G)
  ) / 2

  dimnames(G) <- list(
    ids,
    ids
  )

  eigenvalues_G <- eigen(
    G,
    symmetric = TRUE,
    only.values = TRUE
  )$values

  cv_objects[[fold]] <- list(
    training_index = training_index,
    validation_index = validation_index,
    training_ids = ids[training_index],
    validation_ids = ids[validation_index],
    locus_audit = locus_audit,
    allele_audit = allele_audit,
    marker_means = marker_means,
    allele_frequencies = allele_frequencies,
    denominator = denominator,
    M = M,
    G = G
  )

  fold_diagnostics[[fold]] <- data.frame(
    Fold = fold,
    Training_n = length(training_index),
    Validation_n = length(validation_index),
    Retained_loci = length(retained_loci),
    Retained_alleles = ncol(M),
    Fixed_effect_rank = qr(
      X[
        training_index,
        ,
        drop = FALSE
      ]
    )$rank,
    Mean_diagonal_G = mean(
      diag(G)
    ),
    Minimum_eigenvalue_G = min(
      eigenvalues_G
    )
  )
}
