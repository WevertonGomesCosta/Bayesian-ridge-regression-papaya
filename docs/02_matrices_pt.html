1. Objetivo

Os modelos de predição genômica utilizam matrizes numéricas que representam os efeitos fixos e a semelhança genética entre os indivíduos.

Nesta etapa, os genótipos SSR são convertidos em dosagens alélicas, os efeitos fixos de bloco e família são representados em uma matriz de delineamento e cinco folds de validação cruzada são definidos. O processamento dos marcadores é repetido utilizando apenas os indivíduos de treinamento de cada fold.

3. Codificação das dosagens alélicas

Cada código SSR contém dois rótulos alélicos. O código 12, por exemplo, representa uma cópia do alelo 1 e uma cópia do alelo 2. O código 22 representa duas cópias do alelo 2.

5. Matriz completa de marcadores

As dosagens ausentes são substituídas pela média da coluna. Em seguida, cada média é subtraída de sua respectiva coluna.

A matriz \(\mathbf{M}\) resultante possui os indivíduos nas linhas e as dosagens alélicas mantidas nas colunas.

marker_means_full <- colMeans(
  dosage_full,
  na.rm = TRUE
)

dosage_full_imputed <- dosage_full

for (j in seq_len(ncol(dosage_full_imputed))) {
  missing_values <- is.na(
    dosage_full_imputed[, j]
  )

  dosage_full_imputed[
    missing_values,
    j
  ] <- marker_means_full[j]
}

M_full <- sweep(
  dosage_full_imputed,
  2,
  marker_means_full,
  FUN = "-"
)

6. Matriz completa de relacionamento genômico

Seja \(p_j\) a frequência do alelo representado pela coluna \(j\). A matriz aditiva de relacionamento genômico é calculada por

\[ \mathbf{G} = \frac{\mathbf{M}\mathbf{M}^{\mathsf{T}}} {\sum_j 2p_j(1-p_j)}. \]

allele_frequencies_full <-
  marker_means_full / 2

denominator_full <- sum(
  2 *
    allele_frequencies_full *
    (1 - allele_frequencies_full)
)

G_full <- tcrossprod(
  M_full
) / denominator_full

G_full <- (
  G_full + t(G_full)
) / 2

dimnames(G_full) <- list(
  ids,
  ids
)

eigenvalues_G_full <- eigen(
  G_full,
  symmetric = TRUE,
  only.values = TRUE
)$values

9. Matrizes específicas dos treinamentos

Em cada fold, o filtro dos locos, a codificação dos alelos, o controle das frequências, a imputação, a centralização e a matriz de relacionamento genômico são recalculados com os 120 indivíduos de treinamento.

Os 30 indivíduos de validação são transformados utilizando as médias e a estrutura alélica obtidas no treinamento.

cv_objects <- vector(
  "list",
  N_FOLDS
)

fold_diagnostics <- vector(
  "list",
  N_FOLDS
)

for (fold in seq_len(N_FOLDS)) {
  validation_index <- which(
    folds == fold
  )

  training_index <- which(
    folds != fold
  )

  ssr_training <- ssr_raw[
    training_index,
    ,
    drop = FALSE
  ]

  genotype_classes <- integer(
    ncol(ssr_training)
  )

  for (j in seq_len(ncol(ssr_training))) {
    genotype_classes[j] <- length(
      unique(
        na.omit(
          ssr_training[, j]
        )
      )
    )
  }

  locus_audit <- data.frame(
    Locus = colnames(ssr_training),
    N_called = colSums(!is.na(ssr_training)),
    Call_rate = colMeans(!is.na(ssr_training)),
    N_genotype_classes = genotype_classes
  )

  locus_audit$Keep_locus <-
    locus_audit$Call_rate >= CALL_RATE_MIN &
    locus_audit$N_genotype_classes >= 2

  retained_loci <- locus_audit$Locus[
    locus_audit$Keep_locus
  ]

  ssr_fold <- ssr_raw[
    ,
    retained_loci,
    drop = FALSE
  ]

  dosage_parts <- list()
  allele_map_parts <- list()
  marker_counter <- 0L

  for (j in seq_len(ncol(ssr_fold))) {
    codes_all <- ssr_fold[, j]
    codes_training <- ssr_fold[
      training_index,
      j
    ]

    allele_1_training <- floor(
      codes_training / 10
    )

    allele_2_training <-
      codes_training %% 10

    observed_alleles <- sort(
      unique(
        c(
          allele_1_training[
            !is.na(codes_training)
          ],
          allele_2_training[
            !is.na(codes_training)
          ]
        )
      )
    )

    allele_1_all <- floor(
      codes_all / 10
    )

    allele_2_all <-
      codes_all %% 10

    for (allele in observed_alleles) {
      marker_counter <- marker_counter + 1L

      dosage <-
        as.numeric(
          allele_1_all == allele
        ) +
        as.numeric(
          allele_2_all == allele
        )

      dosage[
        is.na(codes_all)
      ] <- NA_real_

      marker_name <- paste0(
        colnames(ssr_fold)[j],
        "__A",
        allele
      )

      dosage_parts[[marker_counter]] <-
        dosage

      allele_map_parts[[marker_counter]] <-
        data.frame(
          Locus = colnames(ssr_fold)[j],
          Allele = allele,
          Marker = marker_name
        )
    }
  }

  dosage_raw <- do.call(
    cbind,
    dosage_parts
  )

  allele_audit <- bind_rows(
    allele_map_parts
  )

  rownames(dosage_raw) <- ids
  colnames(dosage_raw) <-
    allele_audit$Marker

  allele_audit$Allele_frequency <-
    colMeans(
      dosage_raw[
        training_index,
        ,
        drop = FALSE
      ],
      na.rm = TRUE
    ) / 2

  dosage_sd <- numeric(
    ncol(dosage_raw)
  )

  for (j in seq_len(ncol(dosage_raw))) {
    dosage_sd[j] <- sd(
      dosage_raw[
        training_index,
        j
      ],
      na.rm = TRUE
    )
  }

  allele_audit$Dosage_sd <-
    dosage_sd

  allele_audit$Keep_allele <-
    is.finite(
      allele_audit$Allele_frequency
    ) &
    allele_audit$Allele_frequency >=
      ALLELE_FREQUENCY_MIN &
    allele_audit$Allele_frequency <=
      (1 - ALLELE_FREQUENCY_MIN) &
    is.finite(
      allele_audit$Dosage_sd
    ) &
    allele_audit$Dosage_sd > 0

  retained_markers <-
    allele_audit$Marker[
      allele_audit$Keep_allele
    ]

  dosage <- dosage_raw[
    ,
    retained_markers,
    drop = FALSE
  ]

  marker_means <- colMeans(
    dosage[
      training_index,
      ,
      drop = FALSE
    ],
    na.rm = TRUE
  )

  dosage_imputed <- dosage

  for (j in seq_len(ncol(dosage_imputed))) {
    missing_values <- is.na(
      dosage_imputed[, j]
    )

    dosage_imputed[
      missing_values,
      j
    ] <- marker_means[j]
  }

  M <- sweep(
    dosage_imputed,
    2,
    marker_means,
    FUN = "-"
  )

  allele_frequencies <-
    marker_means / 2

  denominator <- sum(
    2 *
      allele_frequencies *
      (1 - allele_frequencies)
  )

  G <- tcrossprod(
    M
  ) / denominator

  G <- (
    G + t(G)
  ) / 2

  dimnames(G) <- list(
    ids,
    ids
  )

  eigenvalues_G <- eigen(
    G,
    symmetric = TRUE,
    only.values = TRUE
  )$values

  cv_objects[[fold]] <- list(
    training_index = training_index,
    validation_index = validation_index,
    training_ids = ids[training_index],
    validation_ids = ids[validation_index],
    locus_audit = locus_audit,
    allele_audit = allele_audit,
    marker_means = marker_means,
    allele_frequencies = allele_frequencies,
    denominator = denominator,
    M = M,
    G = G
  )

    Fold = fold,

    Validation_n = length(validation_index),

        training_index,
        ,
        drop = FALSE
      ]
    )$rank,
    Mean_diagonal_G = mean(
      diag(G)
    ),
    Minimum_eigenvalue_G = min(
      eigenvalues_G
    )
  )
}
