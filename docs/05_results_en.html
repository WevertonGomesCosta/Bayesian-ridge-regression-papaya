1. Objective

This module presents the results derived in Module 04. No model is refitted, no scientific metric is recalculated, and no selection decision is redefined.

module04_results.rds is the canonical source of analytical results. models_object.rds and native BGLR files are used only for graphical presentation of selected MCMC chains.

Detailed scientific interpretation is reserved for the manuscript.

5. MCMC chains of selected methods

As a visual representation analogous to the reference article, the posterior chain of \(\mu\), its running mean, and posterior density are shown for fold 1 of the selected method for each trait. Complete numerical convergence diagnostics remain in Module 03.

burn_in_saved <- models_object$settings$burn_in / models_object$settings$thin
selected_method_lookup <- setNames(model_selection$Selected_method, model_selection$Trait)

selected_mu_files <- file.path(
  here("output", "models", "bglr"),
  traits,
  "F1",
  unname(selected_method_lookup[traits]),
  "bglr_mu.dat"
)

if (!all(file.exists(selected_mu_files))) {
  stop(
    paste0(
      "M05 integrity gate failed: native fold-1 MCMC files are missing ",
      "for at least one selected method."
    )
  )
}

trace_parts <- vector("list", length(traits))
density_parts <- vector("list", length(traits))

for (trait_index in seq_along(traits)) {
  trait_name <- traits[trait_index]
  method_name <- selected_method_lookup[trait_name]

  mu_file <- file.path(
    here("output", "models", "bglr"),
    trait_name,
    "F1",
    method_name,
    "bglr_mu.dat"
  )

  mu_full <- scan(mu_file, quiet = TRUE)
  posterior_mu <- mu_full[seq.int(burn_in_saved + 1L, length(mu_full))]

  iteration <- seq.int(
    from = models_object$settings$burn_in + models_object$settings$thin,
    by = models_object$settings$thin,
    length.out = length(posterior_mu)
  )

  running_mean <- cumsum(posterior_mu) / seq_along(posterior_mu)
  keep_index <- seq.int(1L, length(posterior_mu), by = 100L)

  trace_parts[[trait_index]] <- data.frame(
    Trait = trait_name,
    Method = method_name,
    Facet = paste0(trait_name, " — ", method_labels[method_name]),
    Iteration = iteration[keep_index],
    Posterior_mu = posterior_mu[keep_index],
    Running_mean = running_mean[keep_index]
  )

  density_object <- density(posterior_mu, n = 512)
  density_parts[[trait_index]] <- data.frame(
    Trait = trait_name,
    Method = method_name,
    Facet = paste0(trait_name, " — ", method_labels[method_name]),
    x = density_object$x,
    y = density_object$y
  )
}

selected_mu_trace <- bind_rows(trace_parts)
selected_mu_density <- bind_rows(density_parts)
